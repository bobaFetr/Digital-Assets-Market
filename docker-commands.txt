# Docker commands for running DAM system on a machine without local .NET/npm installs
# Run these from the project root: c:\Users\Lenovo\Desktop\diplomna\dam

# 1) Build the app image
docker build -f BackEnd/Dockerfile -t dam-app BackEnd

# 2) Create a dedicated Docker network (once)
docker network create dam-net

# 3) The backend uses PostgreSQL through the connection string in
# BackEnd/MyWebApi/appsettings.json. Do not set ConnectionStrings__DefaultConnection
# here unless you intentionally want to override appsettings.json.

# 4) Start the application container
docker run -d --name dam-app --network dam-net -p 8080:10000 -e "Jwt__Key=THIS_IS_A_LONG_RANDOM_SECRET_KEY_32+_CHARS" -e "Jwt__Issuer=MyWebApi" -e "Jwt__Audience=MyWebApiUsers" -e "Frontend__BaseUrl=http://localhost:8080" dam-app

# 5) Check containers are running
docker ps

# 6) View app logs
docker logs -f dam-app

# 7) Open the app and swagger
# App:    http://localhost:8080
# Swagger:http://localhost:8080/swagger

# 8) (Optional) Stop and remove containers
docker stop dam-app
docker rm dam-app

# 9) (Optional) Remove network
docker network rm dam-net
